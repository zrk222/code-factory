class FlowControls {
void direct(javax.servlet.http.HttpServletRequest req, java.sql.Statement db) {
// source: java.request-command-injection
String value = req.getParameter("command");
// ruleid: java.request-command-injection
new java.lang.ProcessBuilder("sh", "-c", value);
// source: java.request-sql-injection
String query = req.getParameter("query");
// ruleid: java.request-sql-injection
db.executeQuery(query);
}
void aliases(javax.servlet.http.HttpServletRequest req, java.sql.Statement db) {
// source: java.request-command-injection
String value = req.getParameter("command");
String alias = "echo " + value;
// ruleid: java.request-command-injection
new java.lang.ProcessBuilder("/bin/sh", "-c", alias);
// source: java.request-sql-injection
String name = req.getParameter("name");
// ruleid: java.parameter-sql-text
String query = "SELECT * FROM users WHERE name='" + name + "'";
// ruleid: java.request-sql-injection
db.execute(query);
// ok: java.request-command-injection
new java.lang.ProcessBuilder("echo", value);
// ok: java.request-command-injection
new java.lang.ProcessBuilder("sh", "-c", "echo fixed");
// ok: java.request-command-injection
new java.lang.ProcessBuilder("/bin/sh", "/opt/fixed-script.sh", value);
// ok: java.request-sql-injection
new java.sql.PreparedStatement("SELECT * FROM users WHERE name=?").setString(1, name);
}
void unrelated(LocalRequest req, LocalStatement db) {
String value = req.getParameter("command");
// ok: java.request-command-injection
new java.lang.ProcessBuilder("sh", "-c", value);
// ok: java.request-sql-injection
db.executeQuery(value);
}
void wrongProvider(javax.servlet.http.HttpServletRequest req, LocalStatement db) {
String value = req.getParameter("query");
// ok: java.request-sql-injection
db.executeQuery(value);
}
void frameworks(jakarta.servlet.http.HttpServletRequest req, java.sql.Connection conn) {
// source: java.request-sql-injection
String name = req.getParameter("name");
// ruleid: java.request-sql-injection, java.parameter-sql-text

// ruleid: java.request-command-injection
Runtime.getRuntime().exec("ping " + host);
}
}

class TypedQuerySinkControls {
  void request(javax.servlet.http.HttpServletRequest req, org.neo4j.driver.Session session,
      org.neo4j.driver.Transaction transaction, org.springframework.jdbc.core.JdbcOperations jdbc) {
    // source: java.request-sql-injection
    String text = req.getParameter("query");
    // ruleid: java.request-sql-injection
    org.neo4j.cypherdsl.core.Cypher.raw(text);
    // source: java.request-sql-injection
    String sessionText = req.getParameter("sessionQuery");
    // ruleid: java.request-sql-injection
    session.run(sessionText);
    // source: java.request-sql-injection
    String transactionText = req.getParameter("transactionQuery");
    // ruleid: java.request-sql-injection
    transaction.run(transactionText);
    // source: java.request-sql-injection
    String jdbcText = req.getParameter("jdbcQuery");
    // ruleid: java.request-sql-injection
    jdbc.queryForList(jdbcText);
    // ok: java.request-sql-injection
    org.neo4j.cypherdsl.core.Cypher.raw("RETURN $E", org.neo4j.cypherdsl.core.Cypher.parameter("value", text));
    // ok: java.request-sql-injection
    session.run("RETURN $value", java.util.Map.of("value", text));
    // ok: java.request-sql-injection
    transaction.run("RETURN $value", java.util.Map.of("value", text));
    // ok: java.request-sql-injection
    jdbc.queryForList("SELECT id FROM records WHERE name = ?", text);
  }
}
