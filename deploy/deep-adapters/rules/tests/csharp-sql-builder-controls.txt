using System.Text.RegularExpressions;

class SqlBuilderControls {
    public void AppendInput(ICommandBuilder builder, string name) {
        var alias = name;
        builder.Append("SELECT * FROM t WHERE name='");
        // ruleid: csharp.request-sql-injection
        builder.Append(alias);
        builder.Append("'");
    }

    public void AppendFixed(ICommandBuilder builder, string unrelated) {
        var fixedValue = "Test";
        builder.Append("SELECT * FROM t WHERE name='");
        // ok: csharp.request-sql-injection
        builder.Append(fixedValue);
        builder.Append("'");
    }

    public void AppendFormatted(ICommandBuilder builder, string name) {
        builder.Append("SELECT * FROM t WHERE name='");
        // ruleid: csharp.request-sql-injection
        builder.Append($"item {name}");
        builder.Append("'");
    }

    public void AppendEscaped(ICommandBuilder builder, string name) {
        builder.Append("SELECT * FROM t WHERE name='");
        // ok: csharp.request-sql-injection
        builder.Append(name.Replace("'", "''"));
        builder.Append("'");
    }

    public void AppendRequest(ICommandBuilder builder, Microsoft.AspNetCore.Http.HttpRequest request) {
        var local = request.Query["name"];
        builder.Append("SELECT * FROM t WHERE name='");
        // ruleid: csharp.request-sql-injection
        builder.Append(local);
        builder.Append("'");
    }
}

internal class SearchProjection : ISqlFragment
{
    private readonly string _configuration;

    public SearchProjection(string configuration = "english")
    {
        _configuration = configuration;
    }

    private string Sql =>
        // ruleid: csharp.request-sql-injection
        $"to_tsquery('{_configuration}'::regconfig, ?)";

    public void Apply(ICommandBuilder builder)
    {
        builder.AppendWithParameters(Sql)[0].Value = "search term";
    }
}

internal class ValidatedSearchProjection : ISqlFragment
{
    private static readonly Regex Identifier = new(
        @"^[a-zA-Z_][a-zA-Z0-9_]{0,62}(\.[a-zA-Z_][a-zA-Z0-9_]{0,62})?$",
        RegexOptions.Compiled | RegexOptions.CultureInvariant);
    private readonly string _configuration;

    public ValidatedSearchProjection(string configuration = "english")
    {
        CheckIdentifier(configuration);
        _configuration = configuration;
    }

    private static void CheckIdentifier(string value)
    {
        if (value is null)
        {
            throw new ArgumentNullException(nameof(value));
        }
        if (!Identifier.IsMatch(value))
        {
            throw new ArgumentException("Invalid search configuration", nameof(value));
        }
    }

    private string Sql =>
        // ok: csharp.request-sql-injection
        $"to_tsquery('{_configuration}'::regconfig, ?)";

    public void Apply(ICommandBuilder builder)
    {
        builder.AppendWithParameters(Sql)[0].Value = "search term";
    }
}
