import { execSync } from "child_process";

const run = (command: string) => execSync(command).toString();

export const dumpDatabase = (database: string) => {
  // ruleid: javascript.parameter-shell-command
  return run(`pg_dump ${database}`);
};

export const listPlugins = (dir: string) => {
  return fs.readdirSync(dir).filter((file) => {
    // ok: javascript.parameter-path-join-without-containment
    return fs.statSync(path.join(dir, file)).isFile();
  });
};
