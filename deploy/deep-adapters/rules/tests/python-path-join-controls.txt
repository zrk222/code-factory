import os
from pathlib import Path

RECIPES = Path("/srv/recipes")


def load_recipe(recipe_name):
    # ruleid: python.parameter-path-join-without-containment
    path = RECIPES / f"{recipe_name}.yaml"
    return path.read_text()


def include(source_file, relpath):
    # ruleid: python.parameter-path-join-without-containment
    dest = os.path.join(os.path.dirname(source_file), relpath)
    with open(dest, "rb") as f:
        return f.read()


def include_checked(source_file, relpath):
    source_dir = os.path.dirname(os.path.realpath(source_file))
    # ok: python.parameter-path-join-without-containment
    dest = os.path.realpath(os.path.join(source_dir, relpath))
    if os.path.commonpath([source_dir, dest]) != source_dir:
        raise ValueError(relpath)
    with open(dest, "rb") as f:
        return f.read()

def _rooted(root, path):
    path.relative_to(root)
    return path


def parent_of(workspace, source, path):
    # A named helper is not a guard the rule can see.
    # ruleid: python.parameter-path-join-without-containment
    parent_file = _rooted(workspace, (source.parent / path).resolve())
    return parent_file.read_text()


def parent_of_enforced(workspace, source, path):
    # ok: python.parameter-path-join-without-containment
    parent_file = (source.parent / path).resolve()
    if not parent_file.is_relative_to(source.parent.resolve()):
        raise ValueError(path)
    return parent_file.read_text()

def save_all(out_dir, pages):
    for page_name, body in pages.items():
        # ok: python.parameter-path-join-without-containment
        target = os.path.abspath(os.path.join(out_dir, page_name))
        if not target.startswith(os.path.abspath(out_dir) + os.sep):
            continue
        with open(target, "w") as handle:
            handle.write(body)


def dump_page(destination, path, content):
    # ok: python.parameter-path-join-without-containment
    path = (Path(destination) / path.lstrip("/")).resolve()
    if not Path(destination) in path.parents:
        raise ValueError(path)
    with open(path, "wb") as handle:
        handle.write(content)


def write_sample(path, filename, payload):
    base_path = os.path.realpath(path)
    # Known gap: a stored result through a second normalized name is a real
    # guard, but the rule only sees a guard in the statement after the join.
    # ruleid: python.parameter-path-join-without-containment
    sample = os.path.join(base_path, filename)
    resolved_sample = os.path.realpath(sample)
    contained = os.path.commonpath((base_path, resolved_sample)) == base_path
    if not contained:
        raise ValueError(filename)
    with open(sample, "w") as handle:
        handle.write(payload)
