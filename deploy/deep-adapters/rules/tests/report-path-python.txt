from pathlib import Path

def write_report(destination, path, content):
    logger.debug(path)
    # ruleid: python.parameter-path-join-without-containment
    output = Path(destination) / path.lstrip("/")
    with open(output, "wb") as stream:
        stream.write(content)

def write_checked(destination, path, content):
    logger.debug(path)
    # ok: python.parameter-path-join-without-containment
    output = Path(destination) / path.lstrip("/")
    output.resolve().relative_to(Path(destination).resolve())
    with open(output, "wb") as stream:
        stream.write(content)

def unrelated_write(destination, path, content):

def read_with_ignored_check(base_dir, filename):
    # ruleid: python.parameter-path-join-without-containment
    target = (base_dir / filename).resolve()
    target.is_relative_to(base_dir.resolve())
    return target.read_text()


def read_with_enforced_check(base_dir, filename):
    # ok: python.parameter-path-join-without-containment
    target = (base_dir / filename).resolve()
    if not target.is_relative_to(base_dir.resolve()):
        raise ValueError("Outside base directory")
    return target.read_text()
