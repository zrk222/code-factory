class CvtModel
  def range(start_time, end_time)
    query = "SELECT * FROM packets "
    # ruleid: ruby.parameter-sql-text
    query += "WHERE PACKET_TIME >= '#{start_time}' AND PACKET_TIME < '#{end_time}'"
    @@conn.exec(query)
  end

  def sorted(query, sort_attr, sort_order)
    # ruleid: ruby.parameter-sql-text
    query.order("#{sort_attr} #{sort_order}")
  end

  def limited(sql, limit)
    # ruleid: ruby.parameter-sql-text
    sql << " LIMIT #{limit}"
  end

  def bound(start_time)
    # ok: ruby.parameter-sql-text
    Packet.where("packet_time >= ?", start_time)
  end

  def ordered(column, direction)
    # ok: ruby.parameter-sql-text
    Packet.order("`#{column}` ASC")
  end

  def by_table(table, id)
    # ok: ruby.parameter-sql-text
    connection.select_all("SELECT * FROM \"#{table}\" WHERE id = $1", "rows", [id])
  end

  def by_table_and_value(table, name)
    # ruleid: ruby.parameter-sql-text
    connection.select_all("SELECT * FROM `#{table}` WHERE name = '#{name}'")
  end

  def quoted(sql, name)
    # ok: ruby.parameter-sql-text
    sql << " WHERE name = #{connection.quote(name)}"
  end

  def counted(sql, limit)
    # ok: ruby.parameter-sql-text
    sql << " LIMIT #{limit.to_i}"
  end

  def message(limit)
    # ok: ruby.parameter-sql-text
    raise ArgumentError, "Body size exceeds limit of #{limit}"
  end

  def cached(account)
    # ok: ruby.parameter-sql-text
    redis.exists?("subscribed:timeline:#{account}")
  end
end

def line_validated_literal(value)
  case value
  # ruleid: ruby.parameter-sql-text
  when /^[01]+$/ then "B'#{value}'"
  # ruleid: ruby.parameter-sql-text
  when /^[0-9A-F]+$/i then "X'#{value}'"
  end
end

def fully_validated_literal(value)
  case value
  # ok: ruby.parameter-sql-text
  when /\A[01]+\z/ then "B'#{value}'"
  # ok: ruby.parameter-sql-text
  when /\A[0-9A-F]+\z/i then "X'#{value}'"
  end
end

def review_recursive_identifier(attrs, default_name = table_name)
  built = attrs.map do |name, value|
    current_table = default_name
    unless value.is_a?(Hash)
      name = name.to_s
      # ruleid: ruby.parameter-sql-text
      if name.include?('.')
        selected, name = name.split('.', 2)
        selected = adapter.quote_table_name(selected)
      else
        selected = current_table
      end
      compare_column("#{selected}.#{adapter.quote_column_name(name)}", value)
    else
      review_recursive_identifier(value, adapter.quote_table_name(name.to_s))
    end
  end.join(' AND ')
  finalize(built)
end

def guarded_recursive_identifier(attrs, default_name = table_name, top_level = true)
  built = attrs.map do |name, value|
    current_table = default_name
    unless value.is_a?(Hash)
      name = name.to_s
      # ok: ruby.parameter-sql-text
      if name.include?('.') && top_level

    else

    end
  end.join(' AND ')
  finalize(built)
end
