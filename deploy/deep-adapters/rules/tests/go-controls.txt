package controls

import (
	"crypto/tls"
	"crypto/x509"
	"net/http"
)

func fetchWithToken(url string, token string) (*http.Response, error) {
	client := &http.Client{Transport: &http.Transport{
		// ruleid: go.disabled-tls-verification
		TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
	}}
	request, _ := http.NewRequest("GET", url, nil)
	request.Header.Set("Authorization", "Bearer "+token)
	return client.Do(request)
}

func fetchWithAssignedConfig(url string, password string) (*http.Response, error) {
	config := &tls.Config{}
	// ruleid: go.disabled-tls-verification
	config.InsecureSkipVerify = true
	request, _ := http.NewRequest("GET", url, nil)
	request.SetBasicAuth("admin", password)
	return (&http.Client{Transport: &http.Transport{TLSClientConfig: config}}).Do(request)
}

func controlPlaneStatus(url string, bearerToken string) (*http.Response, error) {
	client := &http.Client{Transport: &http.Transport{
		// ruleid: go.disabled-tls-verification
		TLSClientConfig: &tls.Config{InsecureSkipVerify: true},
	}}
	request, err := http.NewRequest("GET", url, nil)
	if err != nil {
		return nil, err
	}
	request.Header.Set("Authorization", "Bearer "+bearerToken)
	return client.Do(request)
}

func verifiedControlPlaneStatus(url string, roots *x509.CertPool) (*http.Response, error) {
	client := &http.Client{Transport: &http.Transport{
		// ok: go.disabled-tls-verification
		TLSClientConfig: &tls.Config{RootCAs: roots, MinVersion: tls.VersionTLS12},
	}}
	return client.Get(url)
}

type Probe struct{}

func (p Probe) Handshake(address string) error {
	// ok: go.disabled-tls-verification
	connection, err := tls.Dial("tcp", address, &tls.Config{InsecureSkipVerify: true})
	if err != nil {
		return err
	}
	return connection.Close()
}

func (p Probe) Login(address string, password string) error {
	// ruleid: go.disabled-tls-verification
	connection, err := tls.Dial("tcp", address, &tls.Config{InsecureSkipVerify: true})
	if err != nil {
		return err
	}
	_, err = connection.Write([]byte("AUTH " + password))
	return err
}

type Options struct{ insecure bool }

func fetchOptIn(o Options, url string, token string) (*http.Response, error) {
	client := &http.Client{}
	if o.insecure {
		// ok: go.disabled-tls-verification
		client.Transport = &http.Transport{TLSClientConfig: &tls.Config{InsecureSkipVerify: true}}
	}
	request, _ := http.NewRequest("GET", url, nil)
	request.Header.Set("Authorization", "Bearer "+token)
	return client.Do(request)
}
