export const deleteUpload = async (staticPath: string, doc: { filename: string }) => {
  const fileToDelete = `${staticPath}/${doc.filename}`;
  // ruleid: javascript.parameter-path-join-without-containment
  await fs.unlink(fileToDelete);
};
