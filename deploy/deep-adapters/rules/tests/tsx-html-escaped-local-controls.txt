export function attachment(name: string, href: string) {
  const label = escapeHtml(name);
  // ok: javascript.html-template-unescaped-interpolation
  return `<a class="attachment" download="${label}">${label}</a>`;
}

const sanitizeElement = function (node) {
  const name = lowercase(node.nodeName);
  // ruleid: javascript.dom-innerhtml-from-variable
  if ((name === 'svg' || name === 'math') && node.querySelectorAll('p, form, br').length !== 0) {
    discard(node);
    return true;
  }
};

const sanitizeCoveredElement = function (node) {
  const name = lowercase(node.nodeName);
  // ok: javascript.dom-innerhtml-from-variable
  if ((name === 'svg' || name === 'math') && node.querySelectorAll('p, form, br, table').length !== 0) {
    discard(node);
    return true;
  }
};
