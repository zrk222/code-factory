// Quoted SQL identifiers, identifier escaping helpers and placeholder lists are not values.
export async function quotedIdentifiers(db: any, tableName: string, column: string, id: string) {
  // ok: javascript.parameter-sql-text
  return db.query(`SELECT * FROM "${tableName}" WHERE "${column}" = $1`, [id]);
}

export function schemaQualified(schema: string, table: string) {
  // ok: javascript.parameter-sql-text
  return `SELECT * FROM "${schema}"."${table}" WHERE id = $1`;
}

export function concatenatedIdentifier(table: string) {
  // ok: javascript.parameter-sql-text
  return 'SELECT * FROM "' + table + '" WHERE id = $1';
}

export function escapedIdentifier(table: string, key: string) {
  // ok: javascript.parameter-sql-text
  return `DELETE FROM ${escapeIdentifier(table)} WHERE ${quoteIdent(key)} = $1`;
}

export async function positionalList(db: any, ids: string[]) {
  // ok: javascript.parameter-sql-text
  return db.query(`SELECT * FROM t WHERE id IN (${ids.map((_, i) => `$${i + 1}`).join(', ')})`, ids);
}

export function positionalOffsets(offset: number) {
  // ok: javascript.parameter-sql-text
  return `SELECT * FROM t WHERE a = $${offset + 1} AND b = $${offset + 2}`;
}

export function placeholderLists(placeholders: string[], rows: string[]) {
  // ok: javascript.parameter-sql-text
  const one = `SELECT * FROM t WHERE id IN (${placeholders.join(', ')})`;
  // ok: javascript.parameter-sql-text
  const many = `INSERT INTO t (a) VALUES ${rows.map(() => '(?)').join(', ')}`;
  return [one, many];
}

export async function quotedValue(db: any, name: string) {
  // ruleid: javascript.parameter-sql-text
  return db.query(`SELECT * FROM t WHERE name = '${name}'`);
}

export function doubleQuotedComparison(name: string, pattern: string) {
  // ruleid: javascript.parameter-sql-text
  const equal = `SELECT * FROM t WHERE name = "${name}"`;
  // ruleid: javascript.parameter-sql-text
  const like = `SELECT * FROM t WHERE name LIKE "${pattern}"`;
  return [equal, like];
}

export function quotedValueList(ids: string[]) {
  // ruleid: javascript.parameter-sql-text
  return `SELECT * FROM t WHERE id IN (${ids.map((id) => "'" + id + "'").join(", ")})`;
}

export function identifierAndValue(table: string, id: string) {
  // ruleid: javascript.parameter-sql-text
  return 'SELECT * FROM "' + table + '" WHERE id = ' + id;
}

// A local alias of an identifier quoting helper quotes like the helper itself.
export function aliasedQuoting(options: { db: any; nodeIds: string[]; key: string }) {
  const { db, nodeIds, key } = options;
  const qi = db.sequelize.getQueryInterface();
  const q = qi.quoteIdentifier.bind(qi);
  // ok: javascript.parameter-sql-text
  const bound = `SELECT ${q(key)} FROM t WHERE ${q(key)} IN ($1)`;
  // ruleid: javascript.parameter-sql-text
  const values = `SELECT ${q(key)} FROM t WHERE ${q(key)} IN ('${nodeIds.join("','")}')`;
  return [bound, values];
}

export function arrowAlias(qi: any, key: string, value: string) {
  const q = (name: string) => qi.quoteIdentifier(name);
  const e = qi.escapeId;
  // ok: javascript.parameter-sql-text
  const ids = `SELECT * FROM t WHERE ${q(key)} = $1 AND ${e(key)} = $2`;
  // ruleid: javascript.parameter-sql-text
  const named = `SELECT * FROM t WHERE ${q(key)} = '${value}'`;
  return [ids, named];
}

export function notAQuoter(qi: any, key: string) {
  const f = qi.format.bind(qi);
  // ruleid: javascript.parameter-sql-text
  return `SELECT * FROM t WHERE ${f(key)} = 1`;
}
