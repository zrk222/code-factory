use boring::ssl::SslStream;

fn configure_typed(mut config: boring::ssl::ConnectConfiguration, allow_invalid_hostnames: bool) {
    // ruleid: rust.reqwest-disabled-tls-verification
    config.set_verify_hostname(allow_invalid_hostnames);
}

async fn configure_async(mut config: ProviderAlias, allow_invalid_hostnames: bool, host: &str, stream: Stream) {
    // ruleid: rust.reqwest-disabled-tls-verification
    config.set_verify_hostname(allow_invalid_hostnames);
    let connection = tokio_boring::connect(config, host, stream).await;
}

fn configure_sync(connector: Connector, allow_invalid_hostnames: bool, host: &str, stream: Stream) {
    let connection = connector
        .configure()
        .map_err(convert_error)?
        // ruleid: rust.reqwest-disabled-tls-verification
        .verify_hostname(allow_invalid_hostnames)
        .connect(host, stream);
}

fn configure_project(mut config: LocalConfiguration, allow_invalid_hostnames: bool) {
    // ok: rust.reqwest-disabled-tls-verification
    config.set_verify_hostname(allow_invalid_hostnames);
    // ok: rust.reqwest-disabled-tls-verification
    config.verify_hostname(allow_invalid_hostnames);
}

fn explicit_peer_verification(mut config: boring::ssl::ConnectConfiguration, allow_invalid_hostnames: bool) {
    // ok: rust.reqwest-disabled-tls-verification
    config.set_verify_hostname(!allow_invalid_hostnames);
}
