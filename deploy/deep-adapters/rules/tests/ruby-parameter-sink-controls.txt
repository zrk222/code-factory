class PluginModel
  def install(pypi_url, gem_path)
    pip_args = "-i #{pypi_url} #{gem_path}"
    # ruleid: ruby.parameter-shell-command
    system("pip install #{pip_args}")
  end

  def install_argv(pypi_url, gem_path)
    # ok: ruby.parameter-shell-command
    system("pip", "install", "-i", pypi_url, gem_path)
  end

  def run(cmd)
    # ok: ruby.parameter-shell-command
    system(cmd)
  end

  def open_name(name)
    # ruleid: ruby.parameter-shell-command
    Kernel.open(name)
  end

  def open_asset(name)
    # ok: ruby.parameter-shell-command
    Kernel.open(File.join("assets", name))
  end

  def open_below(directory, name)
    # ruleid: ruby.parameter-shell-command
    Kernel.open(File.join(directory, name))
  end

  def explicit_opt_in(opts, args)
    if args.insecure
      # ok: ruby.disabled-tls-verification
      opts[:verify_ssl] = OpenSSL::SSL::VERIFY_NONE
    end
  end

  def inverted_opt_in(opts, args)
    if !args.insecure
      # ruleid: ruby.disabled-tls-verification
      opts[:verify_ssl] = OpenSSL::SSL::VERIFY_NONE
    end
  end

  def named_option(opts)
    if opts[:skip_tls_verify]
      # ok: ruby.disabled-tls-verification
      opts[:verify_ssl] = OpenSSL::SSL::VERIFY_NONE
    end
  end

  def distinct_option(opts)
    if opts["INSECURE"]
      # ruleid: ruby.disabled-tls-verification
      opts[:verify_ssl] = OpenSSL::SSL::VERIFY_NONE
    end
  end

  def alternative_branch(opts, args)
    if args.insecure
      configure_ok(opts)
    else
      # ruleid: ruby.disabled-tls-verification
      opts[:verify_ssl] = OpenSSL::SSL::VERIFY_NONE
    end
  end
end

def open_explicit_path(path)
  # ruleid: ruby.parameter-shell-command
  Kernel.open(path, "r")
end

def read_file_only(path)
  # ok: ruby.parameter-shell-command
  File.open(path, "r")
  # ok: ruby.parameter-shell-command
  File.read(path)
end

def open_constant_path
  # ok: ruby.parameter-shell-command
  Kernel.open("settings.txt", "r")
end

def open_application_resource(path, store)
  # ok: ruby.parameter-shell-command
  store.open(path, "r")
end

module ArchiveLoader
  def self.open_path(path, mode = "r")
    # ruleid: ruby.parameter-shell-command
    Kernel.open(path, mode)
  end
end
