class PluginModel
  def install(pypi_url, gem_path)
    pip_args = "-i #{pypi_url} #{gem_path}"
    # ruleid: ruby.parameter-shell-command
    system("pip install #{pip_args}")
  end

  def install_argv(pypi_url, gem_path)
    # ok: ruby.parameter-shell-command
    system("pip", "install", "-i", pypi_url, gem_path)
  end

  def run(cmd)
    # ok: ruby.parameter-shell-command
    system(cmd)
  end
end

def open_explicit_path(path)
  # ruleid: ruby.parameter-shell-command
  Kernel.open(path, "r")
end

def read_file_only(path)
  # ok: ruby.parameter-shell-command
  File.open(path, "r")
  # ok: ruby.parameter-shell-command
  File.read(path)
end

def open_constant_path
  # ok: ruby.parameter-shell-command
  Kernel.open("settings.txt", "r")
end

def open_application_resource(path, store)
  # ok: ruby.parameter-shell-command
  store.open(path, "r")
end

module ArchiveLoader
  def self.open_path(path, mode = "r")
    # ruleid: ruby.parameter-shell-command
    Kernel.open(path, mode)
  end
end
